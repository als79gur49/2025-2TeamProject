┌─────────────────────────────────────────────────────────────────────────────┐
│                          INITIALIZATION FLOW                                 │
└─────────────────────────────────────────────────────────────────────────────┘

GameInitializer
    │
    ├──► RegisterGridServices()
    │       └──► GridManager + GridState registered
    │
    ├──► RegisterTeamConfigurationServices()
    │       └──► TeamConfigurationManager registered
    │
    └──► RegisterBaseServices()
            │
            ├──► baseManager.InjectDependencies(gridManager, teamConfigManager)
            └──► ServiceLocator.Register<IBaseManager>(baseManager)


┌─────────────────────────────────────────────────────────────────────────────┐
│                          BASE CREATION FLOW                                  │
└─────────────────────────────────────────────────────────────────────────────┘

GameServiceManager.Start()
    │
    └──► BaseManager.InitializeBases()
            │
            ├──► CreateBase(TeamType.Player, leftPos)
            │      │
            │      ├──► Instantiate(basePrefab)
            │      ├──► base.Initialize(pos, size, team)
            │      ├──► GridState.PlaceBase(baseObj, pos, size, team)
            │      ├──► ApplyTeamVisuals(base, team)
            │      └──► base.OnDeath += HandlePlayerBaseDeath
            │
            └──► CreateBase(TeamType.Enemy, rightPos)
                   │
                   ├──► Instantiate(basePrefab)
                   ├──► base.Initialize(pos, size, team)
                   ├──► GridState.PlaceBase(baseObj, pos, size, team)
                   ├──► ApplyTeamVisuals(base, team)
                   └──► base.OnDeath += HandleEnemyBaseDeath


┌─────────────────────────────────────────────────────────────────────────────┐
│                       BASE DESTRUCTION EVENT FLOW                            │
└─────────────────────────────────────────────────────────────────────────────┘

Enemy Card/Unit attacks Player Base
    │
    ▼
HealthComponent.TakeDamage()
    │
    ▼
Health reaches 0
    │
    ▼
HealthComponent.OnDeath event fires
    │
    ▼
Base.OnBaseDestroyed() [private]
    │
    ├──► Base.OnDeath?.Invoke(gameObject) [public event]
    └──► Cleanup tiles and Destroy(gameObject, 1f)
            │
            ▼
BaseManager.HandlePlayerBaseDeath()
    │
    ├──► Log: "Player Base destroyed"
    └──► OnPlayerBaseDestroyed?.Invoke()
            │
            ▼
GameServiceManager.HandlePlayerBaseDestroyed()
    │
    ├──► Log: "💀 Player Base destroyed - Game Loss"
    ├──► OnPlayerBaseDestroyed?.Invoke() [public event]
    └──► gameService.EndGame()
            │
            ▼
GameService.EndGame()
    │
    ├──► IsGameActive = false
    └──► OnGameEnded?.Invoke()
            │
            ▼
UI shows "Game Over - Defeat"
