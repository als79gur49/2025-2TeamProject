# 개별 인터페이스 기반 유닛 시스템 설계 (방식 B)

## 📋 개요

이 문서는 Unity 게임 개발에서 Health, Attack, MovementRange에 대한 개별 특화 인터페이스를 구현하는 방식(방식 B)에 대한 상세 설계를 다룹니다. 이 접근법은 모듈성, 확장성, 그리고 SOLID 원칙 준수를 중시하는 설계 철학을 기반으로 합니다.

## 🎯 핵심 설계 원칙

### 1. 단일 책임 원칙 (Single Responsibility Principle)
- 각 인터페이스는 하나의 명확한 책임만 가짐
- 시스템 간 의존성 최소화
- 독립적인 기능 개발 및 테스트 가능

### 2. 인터페이스 분리 원칙 (Interface Segregation Principle)
- 클라이언트가 사용하지 않는 메서드에 의존하지 않음
- 필요한 기능만 선택적으로 구현
- 불필요한 구현 강제 방지

### 3. 개방-폐쇄 원칙 (Open-Closed Principle)
- 새로운 시스템 추가 시 기존 코드 수정 불필요
- 확장에는 열려있고 수정에는 닫혀있는 구조

## 🏗️ 인터페이스 구조 설계

### 1. IHealthSystem - 체력 관리 시스템

```csharp
using UnityEngine;

namespace Game.Interfaces
{
    /// <summary>
    /// 체력 관리 시스템 인터페이스
    /// 체력, 데미지, 치료에 관련된 모든 기능을 담당
    /// </summary>
    public interface IHealthSystem
    {
        // 기본 체력 정보
        int MaxHealth { get; }
        int CurrentHealth { get; }
        bool IsAlive { get; }
        
        // 체력 변경 이벤트
        event System.Action<int, int> OnHealthChanged; // (current, max)
        event System.Action OnDeath;
        
        // 체력 조작 메서드
        void TakeDamage(int damage);
        void Heal(int amount);
        void SetMaxHealth(int newMaxHealth);
        
        // 상태 확인
        float HealthPercentage { get; }
        bool IsFullHealth { get; }
        bool IsCriticalHealth { get; } // 체력 25% 미만
    }
}
```

### 2. ICombatSystem - 전투 시스템

```csharp
using UnityEngine;

namespace Game.Interfaces
{
    /// <summary>
    /// 전투 시스템 인터페이스
    /// 공격력, 방어력, 전투 관련 기능을 담당
    /// </summary>
    public interface ICombatSystem
    {
        // 기본 전투 스탯
        int AttackPower { get; }
        int Armor { get; }
        float CriticalChance { get; }
        float CriticalMultiplier { get; }
        
        // 전투 이벤트
        event System.Action<ICombatSystem, int> OnAttackDealt; // (target, damage)
        event System.Action<int, bool> OnDamageDealt; // (damage, isCritical)
        
        // 전투 메서드
        void AttackTarget(IHealthSystem target);
        int CalculateDamage(int baseDamage, int targetArmor);
        bool RollCriticalHit();
        
        // 전투 상태
        bool CanAttack { get; }
        float AttackCooldown { get; }
        void StartAttackCooldown(float duration);
    }
}
```

### 3. IMovementSystem - 이동 시스템

```csharp
using UnityEngine;

namespace Game.Interfaces
{
    /// <summary>
    /// 이동 시스템 인터페이스
    /// 이동력, 위치, 이동 관련 기능을 담당
    /// </summary>
    public interface IMovementSystem
    {
        // 기본 이동 정보
        int MovementRange { get; }
        int ActionPointsPerTurn { get; }
        int CurrentActionPoints { get; }
        Vector2Int Position { get; }
        
        // 이동 이벤트
        event System.Action<Vector2Int, Vector2Int> OnPositionChanged; // (from, to)
        event System.Action<int> OnActionPointsChanged;
        
        // 이동 메서드
        bool CanMoveTo(Vector2Int targetPosition);
        bool MoveTo(Vector2Int targetPosition);
        void SetPosition(Vector2Int newPosition);
        
        // 액션 포인트 관리
        void ConsumeActionPoints(int points);
        void RestoreActionPoints(int points);
        void ResetActionPoints(); // 턴 시작 시 호출
        
        // 이동 계산
        int CalculateMovementCost(Vector2Int from, Vector2Int to);
        bool IsInMovementRange(Vector2Int targetPosition);
    }
}
```

### 4. IBuffSystem - 버프/디버프 시스템 (확장)

```csharp
using System.Collections.Generic;

namespace Game.Interfaces
{
    /// <summary>
    /// 버프/디버프 시스템 인터페이스
    /// 임시 스탯 수정 효과를 담당
    /// </summary>
    public interface IBuffSystem
    {
        // 버프 관리
        IReadOnlyList<IBuff> ActiveBuffs { get; }
        
        // 버프 이벤트
        event System.Action<IBuff> OnBuffApplied;
        event System.Action<IBuff> OnBuffRemoved;
        event System.Action<IBuff> OnBuffExpired;
        
        // 버프 조작
        void ApplyBuff(IBuff buff);
        void RemoveBuff(IBuff buff);
        void RemoveBuffsByType<T>() where T : IBuff;
        void ClearAllBuffs();
        
        // 버프 상태 확인
        bool HasBuff<T>() where T : IBuff;
        T GetBuff<T>() where T : IBuff;
        List<T> GetBuffs<T>() where T : IBuff;
    }
    
    /// <summary>
    /// 개별 버프 인터페이스
    /// </summary>
    public interface IBuff
    {
        string Name { get; }
        string Description { get; }
        float Duration { get; }
        bool IsPermanent { get; }
        bool IsExpired { get; }
        
        void Apply(object target);
        void Remove(object target);
        void Update(float deltaTime);
    }
}
```

## 🔧 구체적인 구현 예시

### 1. 기본 유닛 구현

```csharp
using UnityEngine;
using Game.Interfaces;
using Game.Data;
using System.Collections.Generic;

namespace Game
{
    /// <summary>
    /// 모든 시스템을 구현하는 표준 유닛 클래스
    /// </summary>
    public class StandardUnit : MonoBehaviour, IHealthSystem, ICombatSystem, IMovementSystem, IBuffSystem
    {
        [Header("Unit Data")]
        [SerializeField] private UnitData unitData;
        
        // 런타임 상태
        private int currentHealth;
        private int currentActionPoints;
        private Vector2Int currentPosition;
        private List<IBuff> activeBuffs = new List<IBuff>();
        private float attackCooldownTimer;
        
        #region IHealthSystem Implementation
        
        public int MaxHealth => GetModifiedStat(unitData.GetMaxHealth(), StatType.MaxHealth);
        public int CurrentHealth => currentHealth;
        public bool IsAlive => currentHealth > 0;
        public float HealthPercentage => (float)currentHealth / MaxHealth;
        public bool IsFullHealth => currentHealth >= MaxHealth;
        public bool IsCriticalHealth => HealthPercentage < 0.25f;
        
        public event System.Action<int, int> OnHealthChanged;
        public event System.Action OnDeath;
        
        public void TakeDamage(int damage)
        {
            if (!IsAlive) return;
            
            int actualDamage = Mathf.Max(1, damage - GetModifiedStat(unitData.GetArmor(), StatType.Armor));
            currentHealth = Mathf.Max(0, currentHealth - actualDamage);
            
            OnHealthChanged?.Invoke(currentHealth, MaxHealth);
            
            if (!IsAlive)
            {
                OnDeath?.Invoke();
            }
        }
        
        public void Heal(int amount)
        {
            if (!IsAlive) return;
            
            int oldHealth = currentHealth;
            currentHealth = Mathf.Min(MaxHealth, currentHealth + amount);
            
            if (currentHealth != oldHealth)
            {
                OnHealthChanged?.Invoke(currentHealth, MaxHealth);
            }
        }
        
        public void SetMaxHealth(int newMaxHealth)
        {
            float healthRatio = HealthPercentage;
            currentHealth = Mathf.RoundToInt(newMaxHealth * healthRatio);
            OnHealthChanged?.Invoke(currentHealth, MaxHealth);
        }
        
        #endregion
        
        #region ICombatSystem Implementation
        
        public int AttackPower => GetModifiedStat(unitData.GetAttackPower(), StatType.AttackPower);
        public int Armor => GetModifiedStat(unitData.GetArmor(), StatType.Armor);
        public float CriticalChance => GetModifiedStat(unitData.GetCriticalChance(), StatType.CriticalChance);
        public float CriticalMultiplier => 2.0f;
        public bool CanAttack => IsAlive && attackCooldownTimer <= 0f;
        public float AttackCooldown => attackCooldownTimer;
        
        public event System.Action<ICombatSystem, int> OnAttackDealt;
        public event System.Action<int, bool> OnDamageDealt;
        
        public void AttackTarget(IHealthSystem target)
        {
            if (!CanAttack || target == null || !target.IsAlive) return;
            
            bool isCritical = RollCriticalHit();
            int damage = AttackPower;
            
            if (isCritical)
            {
                damage = Mathf.RoundToInt(damage * CriticalMultiplier);
            }
            
            target.TakeDamage(damage);
            
            OnDamageDealt?.Invoke(damage, isCritical);
            OnAttackDealt?.Invoke(this, damage);
            
            StartAttackCooldown(1.0f); // 1초 쿨다운
        }
        
        public int CalculateDamage(int baseDamage, int targetArmor)
        {
            return Mathf.Max(1, baseDamage - targetArmor);
        }
        
        public bool RollCriticalHit()
        {
            return Random.Range(0f, 1f) < CriticalChance;
        }
        
        public void StartAttackCooldown(float duration)
        {
            attackCooldownTimer = duration;
        }
        
        #endregion
        
        #region IMovementSystem Implementation
        
        public int MovementRange => GetModifiedStat(unitData.GetMovementRange(), StatType.MovementRange);
        public int ActionPointsPerTurn => GetModifiedStat(unitData.GetActionPoints(), StatType.ActionPoints);
        public int CurrentActionPoints => currentActionPoints;
        public Vector2Int Position => currentPosition;
        
        public event System.Action<Vector2Int, Vector2Int> OnPositionChanged;
        public event System.Action<int> OnActionPointsChanged;
        
        public bool CanMoveTo(Vector2Int targetPosition)
        {
            if (!IsAlive) return false;
            
            int cost = CalculateMovementCost(currentPosition, targetPosition);
            return cost <= currentActionPoints && IsInMovementRange(targetPosition);
        }
        
        public bool MoveTo(Vector2Int targetPosition)
        {
            if (!CanMoveTo(targetPosition)) return false;
            
            int cost = CalculateMovementCost(currentPosition, targetPosition);
            Vector2Int oldPosition = currentPosition;
            
            currentPosition = targetPosition;
            ConsumeActionPoints(cost);
            
            OnPositionChanged?.Invoke(oldPosition, currentPosition);
            
            return true;
        }
        
        public void SetPosition(Vector2Int newPosition)
        {
            Vector2Int oldPosition = currentPosition;
            currentPosition = newPosition;
            OnPositionChanged?.Invoke(oldPosition, currentPosition);
        }
        
        public void ConsumeActionPoints(int points)
        {
            currentActionPoints = Mathf.Max(0, currentActionPoints - points);
            OnActionPointsChanged?.Invoke(currentActionPoints);
        }
        
        public void RestoreActionPoints(int points)
        {
            currentActionPoints = Mathf.Min(ActionPointsPerTurn, currentActionPoints + points);
            OnActionPointsChanged?.Invoke(currentActionPoints);
        }
        
        public void ResetActionPoints()
        {
            currentActionPoints = ActionPointsPerTurn;
            OnActionPointsChanged?.Invoke(currentActionPoints);
        }
        
        public int CalculateMovementCost(Vector2Int from, Vector2Int to)
        {
            return Mathf.Abs(to.x - from.x) + Mathf.Abs(to.y - from.y); // 맨하탄 거리
        }
        
        public bool IsInMovementRange(Vector2Int targetPosition)
        {
            int distance = CalculateMovementCost(currentPosition, targetPosition);
            return distance <= MovementRange;
        }
        
        #endregion
        
        #region IBuffSystem Implementation
        
        public IReadOnlyList<IBuff> ActiveBuffs => activeBuffs.AsReadOnly();
        
        public event System.Action<IBuff> OnBuffApplied;
        public event System.Action<IBuff> OnBuffRemoved;
        public event System.Action<IBuff> OnBuffExpired;
        
        public void ApplyBuff(IBuff buff)
        {
            if (buff == null) return;
            
            activeBuffs.Add(buff);
            buff.Apply(this);
            OnBuffApplied?.Invoke(buff);
        }
        
        public void RemoveBuff(IBuff buff)
        {
            if (buff == null || !activeBuffs.Contains(buff)) return;
            
            activeBuffs.Remove(buff);
            buff.Remove(this);
            OnBuffRemoved?.Invoke(buff);
        }
        
        public void RemoveBuffsByType<T>() where T : IBuff
        {
            var buffsToRemove = activeBuffs.FindAll(b => b is T);
            foreach (var buff in buffsToRemove)
            {
                RemoveBuff(buff);
            }
        }
        
        public void ClearAllBuffs()
        {
            var buffsToRemove = new List<IBuff>(activeBuffs);
            foreach (var buff in buffsToRemove)
            {
                RemoveBuff(buff);
            }
        }
        
        public bool HasBuff<T>() where T : IBuff
        {
            return activeBuffs.Exists(b => b is T);
        }
        
        public T GetBuff<T>() where T : IBuff
        {
            return (T)activeBuffs.Find(b => b is T);
        }
        
        public List<T> GetBuffs<T>() where T : IBuff
        {
            var result = new List<T>();
            foreach (var buff in activeBuffs)
            {
                if (buff is T typedBuff)
                {
                    result.Add(typedBuff);
                }
            }
            return result;
        }
        
        #endregion
        
        #region Unity Lifecycle
        
        private void Awake()
        {
            InitializeFromUnitData();
        }
        
        private void Update()
        {
            UpdateCooldowns();
            UpdateBuffs();
        }
        
        private void InitializeFromUnitData()
        {
            if (unitData == null)
            {
                Debug.LogError($"UnitData is null for {gameObject.name}");
                return;
            }
            
            currentHealth = MaxHealth;
            currentActionPoints = ActionPointsPerTurn;
        }
        
        private void UpdateCooldowns()
        {
            if (attackCooldownTimer > 0)
            {
                attackCooldownTimer -= Time.deltaTime;
            }
        }
        
        private void UpdateBuffs()
        {
            var expiredBuffs = new List<IBuff>();
            
            foreach (var buff in activeBuffs)
            {
                buff.Update(Time.deltaTime);
                
                if (buff.IsExpired)
                {
                    expiredBuffs.Add(buff);
                }
            }
            
            foreach (var expiredBuff in expiredBuffs)
            {
                RemoveBuff(expiredBuff);
                OnBuffExpired?.Invoke(expiredBuff);
            }
        }
        
        #endregion
        
        #region Helper Methods
        
        private int GetModifiedStat(int baseStat, StatType statType)
        {
            int modifiedStat = baseStat;
            
            foreach (var buff in activeBuffs)
            {
                if (buff is IStatModifierBuff statBuff)
                {
                    modifiedStat = statBuff.ModifyStat(modifiedStat, statType);
                }
            }
            
            return modifiedStat;
        }
        
        private float GetModifiedStat(float baseStat, StatType statType)
        {
            float modifiedStat = baseStat;
            
            foreach (var buff in activeBuffs)
            {
                if (buff is IStatModifierBuff statBuff)
                {
                    modifiedStat = statBuff.ModifyStat(modifiedStat, statType);
                }
            }
            
            return modifiedStat;
        }
        
        #endregion
    }
    
    public enum StatType
    {
        MaxHealth,
        AttackPower,
        Armor,
        CriticalChance,
        MovementRange,
        ActionPoints
    }
    
    public interface IStatModifierBuff : IBuff
    {
        int ModifyStat(int baseStat, StatType statType);
        float ModifyStat(float baseStat, StatType statType);
    }
}
```

## 🎯 장점 요약

### 1. 모듈성
- ✅ 각 시스템이 독립적으로 개발/테스트 가능
- ✅ 시스템 간 의존성 최소화
- ✅ 명확한 책임 분리

### 2. 확장성
- ✅ 새로운 시스템을 기존 코드 수정 없이 추가
- ✅ 기존 시스템의 안정성 보장
- ✅ 점진적 기능 개발 가능

### 3. 재사용성

- ✅ 시스템별 로직 재사용
- ✅ 조합을 통한 다양한 유닛 생성

## 🚀 마이그레이션 전략

### 1. 점진적 도입
1. 새로운 인터페이스 정의
2. 기존 Unit 클래스에 인터페이스 구현 추가
3. 새로운 기능은 인터페이스 기반으로 개발
4. 레거시 코드 점진적 리팩토링

### 2. 하이브리드 접근

```csharp
    // 기존 코드가 사용할 수 있는 통합 인터페이스
}
```
